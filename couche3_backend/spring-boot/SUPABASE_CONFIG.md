## 📋 Structure finale de `evenements_bruts`

```sql
CREATE TABLE IF NOT EXISTS evenements_bruts (
    id          BIGINT PRIMARY KEY GENERATED BY DEFAULT AS IDENTITY,
    conducteur_id    VARCHAR(20) NOT NULL,           -- Format: C{digits}
    type_evenement   VARCHAR(50) NOT NULL,           -- FATIGUE_DROP, PHONE, etc.
    categorie        VARCHAR(10) NOT NULL,           -- 'DMS' ou 'ADAS'
    timestamp_utc    TIMESTAMP NOT NULL DEFAULT NOW(), -- UTC
    duree_secondes   DECIMAL(8,2) NOT NULL DEFAULT 0,  -- Durée événement
    latitude         DECIMAL(10,7),                   -- Position GPS
    longitude        DECIMAL(10,7),                  -- Position GPS
    severite         VARCHAR(10) NOT NULL DEFAULT 'MODERE', 
                     -- 'FAIBLE', 'MODERE', 'ELEVE', 'CRITIQUE'
    traite           BOOLEAN DEFAULT FALSE,          -- ✓ IMPORTANT: évite doublon
    crc              VARCHAR(255),                   -- Intégrité données
    created_at       TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_traite ON evenements_bruts(traite);
CREATE INDEX idx_conducteur ON evenements_bruts(conducteur_id);
```

## 🔐 Configuration RLS (Row Level Security)

```sql

-- 1️⃣ Désactiver RLS sur evenements_bruts (READ-ONLY from Python & Spring Boot)
ALTER TABLE evenements_bruts DISABLE ROW LEVEL SECURITY;

-- 2️⃣ Donner accès complet à anon/authenticated
GRANT SELECT, UPDATE(traite) ON evenements_bruts TO anon, authenticated;

-- 3️⃣ Vérifier
SELECT tablename FROM pg_tables 
WHERE schemaname = 'public' 
AND tablename = 'evenements_bruts';
```
